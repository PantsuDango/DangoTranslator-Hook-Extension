cmake_minimum_required(VERSION 3.16)
project(DangoUnityScenario LANGUAGES CXX)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Unity Scenario requires x64 Mono")
endif()
add_executable(UnityScenarioHost ../livemaker/main.cpp)
add_executable(UnityScenarioInjector ../livemaker/injector.cpp)
target_compile_features(UnityScenarioHost PRIVATE cxx_std_17)
target_compile_features(UnityScenarioInjector PRIVATE cxx_std_17)
target_compile_definitions(UnityScenarioHost PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX DANGO_UNITY_SCENARIO_ONLY=1)
target_compile_definitions(UnityScenarioInjector PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
foreach(target UnityScenarioHost UnityScenarioInjector)
    if(MSVC)
        target_compile_options(${target} PRIVATE /utf-8 /W4 /EHsc)
        set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
    set_target_properties(${target} PROPERTIES OUTPUT_NAME "${target}64" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endforeach()
