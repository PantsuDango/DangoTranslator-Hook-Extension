cmake_minimum_required(VERSION 3.16)
project(DangoDxCompiledHost LANGUAGES CXX)
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "DxCompiled requires Win32")
endif()
# Reuse the bounded native snapshot host protocol, with an independent engine,
# hook allow-list, channel magic and resource format. The reference is never edited.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../livemaker/main.cpp" host_source)
string(REPLACE "NSCRIPTER" "DXCOMPILED" host_source "${host_source}")
string(REPLACE "NScripter" "DxCompiled" host_source "${host_source}")
string(REPLACE "nscripter" "dxcompiled" host_source "${host_source}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/DxCompiledHost.cpp" "${host_source}")
add_executable(DxCompiledHost32 "${CMAKE_CURRENT_BINARY_DIR}/DxCompiledHost.cpp")
target_include_directories(DxCompiledHost32 PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../livemaker")
target_compile_definitions(DxCompiledHost32 PRIVATE DANGO_DXCOMPILED_ONLY=1)
add_executable(DxCompiledInjector32 ../livemaker/injector.cpp)
target_compile_definitions(DxCompiledInjector32 PRIVATE DANGO_DXCOMPILED_ONLY=1)
target_link_libraries(DxCompiledInjector32 PRIVATE advapi32)
foreach(target DxCompiledHost32 DxCompiledInjector32)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(${target} PRIVATE /utf-8 /W4 /EHsc)
  set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endforeach()
