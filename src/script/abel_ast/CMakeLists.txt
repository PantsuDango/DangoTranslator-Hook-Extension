cmake_minimum_required(VERSION 3.16)
project(DangoAbelASTHost LANGUAGES CXX)
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "AbelAST requires Win32")
endif()
# Reuse the bounded native snapshot host protocol, with an independent engine,
# hook allow-list, channel magic and resource format. The reference is never edited.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../livemaker/main.cpp" host_source)
string(REPLACE "NSCRIPTER" "ABELAST" host_source "${host_source}")
string(REPLACE "NScripter" "AbelAST" host_source "${host_source}")
string(REPLACE "nscripter" "abelast" host_source "${host_source}")
string(REPLACE "../abelast/" "${CMAKE_CURRENT_SOURCE_DIR}/" host_source "${host_source}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/AbelASTHost.cpp" "${host_source}")
add_executable(AbelASTHost32 "${CMAKE_CURRENT_BINARY_DIR}/AbelASTHost.cpp")
target_include_directories(AbelASTHost32 PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../livemaker")
target_compile_definitions(AbelASTHost32 PRIVATE DANGO_ABELAST_ONLY=1)
add_executable(AbelASTInjector32 ../livemaker/injector.cpp)
foreach(target AbelASTHost32 AbelASTInjector32)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(${target} PRIVATE /utf-8 /W4 /EHsc)
  set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endforeach()
