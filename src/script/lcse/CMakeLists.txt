cmake_minimum_required(VERSION 3.16)
project(DangoLCSEHookHost LANGUAGES CXX)

add_executable(LCSEHost32 ../livemaker/main.cpp)
add_executable(LCSEInjector32 ../livemaker/injector.cpp)
target_compile_features(LCSEHost32 PRIVATE cxx_std_17)
target_compile_features(LCSEInjector32 PRIVATE cxx_std_17)
target_compile_definitions(LCSEHost32 PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX DANGO_LCSE_ONLY=1)
target_compile_definitions(LCSEInjector32 PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)

if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "LCSE Hook 组件必须使用 Win32 工具链构建")
endif()
if(MSVC)
    foreach(target LCSEHost32 LCSEInjector32)
        target_compile_options(${target} PRIVATE /utf-8 /W4 /EHsc)
        set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endforeach()
endif()
set_target_properties(LCSEHost32 LCSEInjector32 PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
