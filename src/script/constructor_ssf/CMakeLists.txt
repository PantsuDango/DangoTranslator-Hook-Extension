cmake_minimum_required(VERSION 3.16)
project(DangoConstructorSSFHost LANGUAGES CXX)
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "ConstructorSSF requires Win32")
endif()
# Reuse the bounded native snapshot host protocol, with an independent engine,
# hook allow-list, channel magic and resource format. The reference is never edited.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../livemaker/main.cpp" host_source)
string(REPLACE "NSCRIPTER" "CONSTRUCTORSSF" host_source "${host_source}")
string(REPLACE "NScripter" "ConstructorSSF" host_source "${host_source}")
string(REPLACE "nscripter" "constructorssf" host_source "${host_source}")
string(REPLACE "../constructorssf/" "${CMAKE_CURRENT_SOURCE_DIR}/" host_source "${host_source}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/ConstructorSSFHost.cpp" "${host_source}")
add_executable(ConstructorSSFHost32 "${CMAKE_CURRENT_BINARY_DIR}/ConstructorSSFHost.cpp")
target_include_directories(ConstructorSSFHost32 PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../livemaker")
target_compile_definitions(ConstructorSSFHost32 PRIVATE DANGO_CONSTRUCTORSSF_ONLY=1)
add_executable(ConstructorSSFInjector32 ../livemaker/injector.cpp)
foreach(target ConstructorSSFHost32 ConstructorSSFInjector32)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(${target} PRIVATE /utf-8 /W4 /EHsc)
  set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endforeach()
option(DANGO_CONSTRUCTORSSF_TESTS "Build native display regression tests" OFF)
if(DANGO_CONSTRUCTORSSF_TESTS)
  enable_testing()
  add_executable(ConstructorSSFDisplayTest tests/display_test.cpp)
  target_compile_features(ConstructorSSFDisplayTest PRIVATE cxx_std_17)
  target_compile_definitions(ConstructorSSFDisplayTest PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(ConstructorSSFDisplayTest PRIVATE /utf-8 /W4 /EHsc)
  target_link_libraries(ConstructorSSFDisplayTest PRIVATE gdi32 user32 advapi32)
  add_test(NAME ConstructorSSFDisplay COMMAND ConstructorSSFDisplayTest)
endif()
