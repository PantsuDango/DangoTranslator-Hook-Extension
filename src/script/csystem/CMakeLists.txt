cmake_minimum_required(VERSION 3.16)
project(DangoCSystemHookHost LANGUAGES CXX)
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "CSystem Hook requires Win32")
endif()
add_executable(CSystemHost32 ../livemaker/main.cpp)
add_executable(CSystemInjector32 ../livemaker/injector.cpp)
foreach(target CSystemHost32 CSystemInjector32)
    target_compile_features(${target} PRIVATE cxx_std_17)
    target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    if(MSVC)
        target_compile_options(${target} PRIVATE /utf-8 /W4 /EHsc)
        set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
    set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endforeach()
target_compile_definitions(CSystemHost32 PRIVATE DANGO_CSYSTEM_ONLY=1)
