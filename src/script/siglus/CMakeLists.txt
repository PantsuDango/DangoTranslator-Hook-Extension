cmake_minimum_required(VERSION 3.16)
project(DangoSiglusHookHost LANGUAGES CXX)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Siglus组件需要Win32目标")
endif()
add_executable(SiglusHost32 ../livemaker/main.cpp)
add_executable(SiglusInjector32 ../livemaker/injector.cpp)
foreach(target SiglusHost32 SiglusInjector32)
    target_compile_features(${target} PRIVATE cxx_std_17)
    target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX DANGO_SIGLUS_ONLY=1)
    if(MSVC)
        target_compile_options(${target} PRIVATE /utf-8 /W4 /EHsc)
        set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
    set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endforeach()
