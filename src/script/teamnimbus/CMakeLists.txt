cmake_minimum_required(VERSION 3.16)
project(DangoTeamNimbus LANGUAGES CXX)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "TeamNimbus requires x64 Mono")
endif()
add_executable(TeamNimbusHost ../livemaker/main.cpp)
add_executable(TeamNimbusInjector ../livemaker/injector.cpp)
foreach(target TeamNimbusHost TeamNimbusInjector)
    target_compile_features(${target} PRIVATE cxx_std_17)
    target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    if(MSVC)
        target_compile_options(${target} PRIVATE /utf-8 /W4 /EHsc)
        set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
    set_target_properties(${target} PROPERTIES OUTPUT_NAME "${target}64" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endforeach()
target_compile_definitions(TeamNimbusHost PRIVATE DANGO_TEAMNIMBUS_ONLY=1)
