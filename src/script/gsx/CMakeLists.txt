cmake_minimum_required(VERSION 3.16)
project(DangoGSXHost LANGUAGES CXX)
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "GSX requires Win32")
endif()
# Reuse the bounded native snapshot host protocol, with an independent engine,
# hook allow-list, channel magic and resource format. The reference is never edited.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../livemaker/main.cpp" host_source)
string(REPLACE "NSCRIPTER" "GSX" host_source "${host_source}")
string(REPLACE "NScripter" "GSX" host_source "${host_source}")
string(REPLACE "nscripter" "gsx" host_source "${host_source}")
string(REPLACE "../gsx/" "${CMAKE_CURRENT_SOURCE_DIR}/" host_source "${host_source}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/GSXHost.cpp" "${host_source}")
add_executable(GSXHost32 "${CMAKE_CURRENT_BINARY_DIR}/GSXHost.cpp")
target_include_directories(GSXHost32 PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../livemaker")
target_compile_definitions(GSXHost32 PRIVATE DANGO_GSX_ONLY=1)
add_executable(GSXInjector32 ../livemaker/injector.cpp)
foreach(target GSXHost32 GSXInjector32)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(${target} PRIVATE /utf-8 /W4 /EHsc)
  set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endforeach()
