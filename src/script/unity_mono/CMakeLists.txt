cmake_minimum_required(VERSION 3.16)
project(DangoUnityMono LANGUAGES CXX)
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    set(BITS 64)
else()
    set(BITS 32)
endif()
add_executable(UnityMonoHost ../livemaker/main.cpp)
add_executable(UnityMonoInjector ../livemaker/injector.cpp)
foreach(target UnityMonoHost UnityMonoInjector)
    target_compile_features(${target} PRIVATE cxx_std_17)
    target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    if(MSVC)
        target_compile_options(${target} PRIVATE /utf-8 /W4 /EHsc)
        set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
    set_target_properties(${target} PROPERTIES OUTPUT_NAME "${target}${BITS}" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endforeach()
# Reuse the existing lease, packet transport and process lifecycle implementation.
target_compile_definitions(UnityMonoHost PRIVATE DANGO_TEAMNIMBUS_ONLY=1 DANGO_UNITY_MONO_ONLY=1)
